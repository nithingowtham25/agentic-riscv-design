module tb_alu;

  logic [31:0] A;
  logic [31:0] B;
  logic [2:0]  ALUSelect;
  logic        SubArith;
  logic [31:0] ALUResult;
  logic [31:0] Sum;

  integer tests_passed;
  integer tests_failed;
  integer tests_total;

  logic signed [31:0] sA;
  logic signed [31:0] sB;

  alu dut (
    .A(A),
    .B(B),
    .ALUSelect(ALUSelect),
    .SubArith(SubArith),
    .ALUResult(ALUResult),
    .Sum(Sum)
  );

  task automatic run_test(
    input [31:0] tA,
    input [31:0] tB,
    input [2:0]  tALUSelect,
    input        tSubArith,
    input [255:0] test_name
  );
    reg [31:0] expected_result;
    reg [31:0] expected_sum;
    begin
      A = tA;
      B = tB;
      ALUSelect = tALUSelect;
      SubArith = tSubArith;

      #1;

      expected_sum = tSubArith ? (tA - tB) : (tA + tB);

      case (tALUSelect)
        3'b000: expected_result = tSubArith ? (tA - tB) : (tA + tB);
        3'b001: expected_result = tA << tB[4:0];
        3'b010: expected_result = ($signed(tA) < $signed(tB)) ? 32'h00000001 : 32'h00000000;
        3'b011: expected_result = (tA < tB) ? 32'h00000001 : 32'h00000000;
        3'b100: expected_result = tA ^ tB;
        3'b101: expected_result = tSubArith ? ($signed(tA) >>> tB[4:0]) : (tA >> tB[4:0]);
        3'b110: expected_result = tA | tB;
        3'b111: expected_result = tA & tB;
        default: expected_result = 32'hxxxxxxxx;
      endcase

      tests_total = tests_total + 1;

      if ((ALUResult === expected_result) && (Sum === expected_sum)) begin
        tests_passed = tests_passed + 1;
      end else begin
        tests_failed = tests_failed + 1;
        $display("FAIL: %0s", test_name);
        $display("  Inputs: A=0x%08h B=0x%08h ALUSelect=%03b SubArith=%0d", tA, tB, tALUSelect, tSubArith);
        $display("  Expected: ALUResult=0x%08h Sum=0x%08h", expected_result, expected_sum);
        $display("  Actual  : ALUResult=0x%08h Sum=0x%08h", ALUResult, Sum);
      end
    end
  endtask

  initial begin
    tests_passed = 0;
    tests_failed = 0;
    tests_total  = 0;

    A = 32'h0;
    B = 32'h0;
    ALUSelect = 3'b000;
    SubArith = 1'b0;

    run_test(32'h0000000A, 32'h00000003, 3'b000, 1'b0, "ADD basic");
    run_test(32'h0000000A, 32'h00000003, 3'b000, 1'b1, "SUB basic");
    run_test(32'hFFFFFFFF, 32'h00000001, 3'b000, 1'b0, "ADD wraparound");
    run_test(32'h00000000, 32'h00000001, 3'b000, 1'b1, "SUB wraparound");

    run_test(32'h0000000F, 32'h0000001F, 3'b001, 1'b0, "SLL by 31");
    run_test(32'h00000001, 32'h00000020, 3'b001, 1'b0, "SLL uses only B[4:0] (32->0)");
    run_test(32'h12345678, 32'h00000004, 3'b001, 1'b1, "SLL ignores SubArith for result but Sum uses subtraction");

    run_test(32'hFFFFFFFF, 32'h00000001, 3'b010, 1'b1, "SLT signed true (-1 < 1)");
    run_test(32'h7FFFFFFF, 32'h80000000, 3'b010, 1'b1, "SLT signed false (max positive < min negative)");
    run_test(32'hFFFFFFFF, 32'h00000001, 3'b011, 1'b1, "SLTU unsigned false");
    run_test(32'h00000000, 32'hFFFFFFFF, 3'b011, 1'b1, "SLTU unsigned true");

    run_test(32'hA5A5A5A5, 32'h5A5A5A5A, 3'b100, 1'b0, "XOR pattern");
    run_test(32'h0F0F0F0F, 32'hF00000FF, 3'b110, 1'b0, "OR pattern");
    run_test(32'hFFFF0000, 32'h0F0FF0F0, 3'b111, 1'b0, "AND pattern");

    run_test(32'h80000000, 32'h00000001, 3'b101, 1'b1, "SRA negative by 1");
    run_test(32'h80000000, 32'h0000001F, 3'b101, 1'b1, "SRA negative by 31");
    run_test(32'h80000000, 32'h00000001, 3'b101, 1'b0, "SRL by 1");
    run_test(32'h12345678, 32'h00000020, 3'b101, 1'b0, "SRL uses only B[4:0] (32->0)");

    run_test(32'h11111111, 32'h22222222, 3'b110, 1'b1, "OR ignores SubArith for result but Sum subtracts");
    run_test(32'h33333333, 32'h11111111, 3'b111, 1'b1, "AND ignores SubArith for result but Sum subtracts");

    $display("TESTS_PASSED: %0d", tests_passed);
    $display("TESTS_FAILED: %0d", tests_failed);
    $display("TESTS_TOTAL: %0d", tests_total);
    if (tests_failed == 0)
      $display("RESULT: PASS");
    else
      $display("RESULT: FAIL");

    $finish;
  end

endmodule
