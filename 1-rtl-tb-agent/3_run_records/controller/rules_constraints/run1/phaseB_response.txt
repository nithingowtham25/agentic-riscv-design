module tb_controller;

  logic [31:0] InstrD;
  logic        RegWriteD;
  logic [2:0]  ImmSrcD;
  logic        ALUSrcAD;
  logic        ALUSrcBD;
  logic [1:0]  MemRWD;
  logic [2:0]  ResultSrcD;
  logic        BranchD;
  logic        JumpD;
  logic        ALUResultSrcD;
  logic [2:0]  ALUSelectD;
  logic        SubArithD;
  logic        IllegalInstrD;

  integer pass_count;
  integer fail_count;
  integer total_count;

  controller dut (
    .InstrD(InstrD),
    .RegWriteD(RegWriteD),
    .ImmSrcD(ImmSrcD),
    .ALUSrcAD(ALUSrcAD),
    .ALUSrcBD(ALUSrcBD),
    .MemRWD(MemRWD),
    .ResultSrcD(ResultSrcD),
    .BranchD(BranchD),
    .JumpD(JumpD),
    .ALUResultSrcD(ALUResultSrcD),
    .ALUSelectD(ALUSelectD),
    .SubArithD(SubArithD),
    .IllegalInstrD(IllegalInstrD)
  );

  task automatic check_case;
    input [255:0] name;
    input [31:0] instr;
    input exp_RegWriteD;
    input [2:0] exp_ImmSrcD;
    input exp_ALUSrcAD;
    input exp_ALUSrcBD;
    input [1:0] exp_MemRWD;
    input [2:0] exp_ResultSrcD;
    input exp_BranchD;
    input exp_JumpD;
    input exp_ALUResultSrcD;
    input [2:0] exp_ALUSelectD;
    input exp_SubArithD;
    input exp_IllegalInstrD;
    begin
      InstrD = instr;
      #1;
      total_count = total_count + 1;

      if ((RegWriteD     !== exp_RegWriteD)     ||
          (ImmSrcD       !== exp_ImmSrcD)       ||
          (ALUSrcAD      !== exp_ALUSrcAD)      ||
          (ALUSrcBD      !== exp_ALUSrcBD)      ||
          (MemRWD        !== exp_MemRWD)        ||
          (ResultSrcD    !== exp_ResultSrcD)    ||
          (BranchD       !== exp_BranchD)       ||
          (JumpD         !== exp_JumpD)         ||
          (ALUResultSrcD !== exp_ALUResultSrcD) ||
          (ALUSelectD    !== exp_ALUSelectD)    ||
          (SubArithD     !== exp_SubArithD)     ||
          (IllegalInstrD !== exp_IllegalInstrD)) begin
        fail_count = fail_count + 1;
        $display("FAIL: %0s instr=%h", name, instr);
        $display("  Expected: RegWrite=%0d ImmSrc=%03b ALUSrcA=%0d ALUSrcB=%0d MemRW=%02b ResultSrc=%03b Branch=%0d Jump=%0d ALUResultSrc=%0d ALUSelect=%03b SubArith=%0d Illegal=%0d",
                 exp_RegWriteD, exp_ImmSrcD, exp_ALUSrcAD, exp_ALUSrcBD, exp_MemRWD, exp_ResultSrcD,
                 exp_BranchD, exp_JumpD, exp_ALUResultSrcD, exp_ALUSelectD, exp_SubArithD, exp_IllegalInstrD);
        $display("  Actual:   RegWrite=%0d ImmSrc=%03b ALUSrcA=%0d ALUSrcB=%0d MemRW=%02b ResultSrc=%03b Branch=%0d Jump=%0d ALUResultSrc=%0d ALUSelect=%03b SubArith=%0d Illegal=%0d",
                 RegWriteD, ImmSrcD, ALUSrcAD, ALUSrcBD, MemRWD, ResultSrcD,
                 BranchD, JumpD, ALUResultSrcD, ALUSelectD, SubArithD, IllegalInstrD);
      end else begin
        pass_count = pass_count + 1;
      end
    end
  endtask

  initial begin
    pass_count = 0;
    fail_count = 0;
    total_count = 0;
    InstrD = 32'h00000000;

    // R-type ADD: opcode=0110011 funct3=000 funct7=0000000
    check_case("R-type ADD",
               32'b0000000_00011_00010_000_00001_0110011,
               1'b1, 3'b000, 1'b0, 1'b0, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b0);

    // R-type SUB: funct7=0100000
    check_case("R-type SUB",
               32'b0100000_00011_00010_000_00001_0110011,
               1'b1, 3'b000, 1'b0, 1'b0, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b1, 1'b0);

    // R-type SRA: funct3=101 funct7=0100000
    check_case("R-type SRA",
               32'b0100000_00011_00010_101_00001_0110011,
               1'b1, 3'b000, 1'b0, 1'b0, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b101, 1'b1, 1'b0);

    // I-type XORI
    check_case("I-type XORI",
               32'b000000000101_00010_100_00001_0010011,
               1'b1, 3'b000, 1'b0, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b100, 1'b0, 1'b0);

    // I-type SLTI
    check_case("I-type SLTI",
               32'b000000000001_00010_010_00001_0010011,
               1'b1, 3'b000, 1'b0, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b010, 1'b1, 1'b0);

    // I-type SRAI legal: funct3=101, imm[11:5]=0100000
    check_case("I-type SRAI legal",
               32'b0100000_00101_00010_101_00001_0010011,
               1'b1, 3'b000, 1'b0, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b101, 1'b1, 1'b0);

    // I-type shift-immediate illegal: funct3=001 with bad imm[11:5]
    check_case("I-type SLLI illegal encoding",
               32'b0100000_00101_00010_001_00001_0010011,
               1'b0, 3'b000, 1'b0, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b001, 1'b0, 1'b1);

    // Load LW legal: funct3=010
    check_case("Load LW",
               32'b000000000000_00010_010_00001_0000011,
               1'b1, 3'b000, 1'b0, 1'b1, 2'b10, 3'b001, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b0);

    // Load illegal funct3
    check_case("Load illegal funct3",
               32'b000000000000_00010_111_00001_0000011,
               1'b0, 3'b000, 1'b0, 1'b1, 2'b00, 3'b001, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b1);

    // Store SW legal: funct3=010
    check_case("Store SW",
               32'b0000000_00001_00010_010_00000_0100011,
               1'b0, 3'b001, 1'b0, 1'b1, 2'b01, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b0);

    // Store illegal funct3
    check_case("Store illegal funct3",
               32'b0000000_00001_00010_011_00000_0100011,
               1'b0, 3'b001, 1'b0, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b1);

    // Branch BEQ legal
    check_case("Branch BEQ",
               32'b0000000_00011_00010_000_00000_1100011,
               1'b0, 3'b010, 1'b1, 1'b1, 2'b00, 3'b000, 1'b1, 1'b0, 1'b0, 3'b000, 1'b0, 1'b0);

    // Branch BLTU legal
    check_case("Branch BLTU",
               32'b0000000_00011_00010_110_00000_1100011,
               1'b0, 3'b010, 1'b1, 1'b1, 2'b00, 3'b000, 1'b1, 1'b0, 1'b0, 3'b000, 1'b0, 1'b0);

    // Branch illegal funct3
    check_case("Branch illegal funct3",
               32'b0000000_00011_00010_010_00000_1100011,
               1'b0, 3'b010, 1'b1, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b1);

    // AUIPC
    check_case("AUIPC",
               32'b00000000000000000000_00001_0010111,
               1'b1, 3'b100, 1'b1, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b0);

    // LUI
    check_case("LUI",
               32'b00000000000000000000_00001_0110111,
               1'b1, 3'b100, 1'b0, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b1, 3'b000, 1'b0, 1'b0);

    // JALR legal
    check_case("JALR",
               32'b000000000000_00010_000_00001_1100111,
               1'b1, 3'b000, 1'b0, 1'b1, 2'b00, 3'b000, 1'b0, 1'b1, 1'b1, 3'b000, 1'b0, 1'b0);

    // JALR illegal funct3
    check_case("JALR illegal funct3",
               32'b000000000000_00010_001_00001_1100111,
               1'b0, 3'b000, 1'b0, 1'b1, 2'b00, 3'b000, 1'b0, 1'b0, 1'b1, 3'b000, 1'b0, 1'b1);

    // JAL
    check_case("JAL",
               32'b00000000000000000000_00001_1101111,
               1'b1, 3'b011, 1'b1, 1'b1, 2'b00, 3'b000, 1'b0, 1'b1, 1'b1, 3'b000, 1'b0, 1'b0);

    // Illegal opcode
    check_case("Illegal opcode",
               32'b00000000000000000000000001111111,
               1'b0, 3'b000, 1'b0, 1'b0, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b1);

    // Illegal extension/system instruction (ECALL encoding)
    check_case("ECALL illegal",
               32'b00000000000000000000000001110011,
               1'b0, 3'b000, 1'b0, 1'b0, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b1);

    // R-type illegal funct7
    check_case("R-type illegal funct7",
               32'b0000001_00011_00010_000_00001_0110011,
               1'b0, 3'b000, 1'b0, 1'b0, 2'b00, 3'b000, 1'b0, 1'b0, 1'b0, 3'b000, 1'b0, 1'b1);

    $display("TESTS_PASSED: %0d", pass_count);
    $display("TESTS_FAILED: %0d", fail_count);
    $display("TESTS_TOTAL: %0d", total_count);
    if (fail_count == 0)
      $display("RESULT: PASS");
    else
      $display("RESULT: FAIL");

    $finish;
  end

endmodule
