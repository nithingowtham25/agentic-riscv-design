```systemverilog
module controller (
    input  logic [31:0] InstrD,
    output logic        RegWriteD,
    output logic [2:0]  ImmSrcD,
    output logic        ALUSrcAD,
    output logic        ALUSrcBD,
    output logic [1:0]  MemRWD,
    output logic [2:0]  ResultSrcD,
    output logic        BranchD,
    output logic        JumpD,
    output logic        ALUResultSrcD,
    output logic [2:0]  ALUSelectD,
    output logic        SubArithD,
    output logic        IllegalInstrD
);

    // -------------------------------------------------------------------------
    // 1. Inputs / outputs / widths
    //    Input:
    //      - InstrD[31:0]
    //    Outputs:
    //      - RegWriteD
    //      - ImmSrcD[2:0]
    //      - ALUSrcAD
    //      - ALUSrcBD
    //      - MemRWD[1:0]
    //      - ResultSrcD[2:0]
    //      - BranchD
    //      - JumpD
    //      - ALUResultSrcD
    //      - ALUSelectD[2:0]
    //      - SubArithD
    //      - IllegalInstrD
    //
    // 2. Required operations / control conditions
    //    - Decode RV32I instruction classes from opcode/funct3/funct7.
    //    - Produce combinational datapath control signals.
    //    - Enforce strict legality:
    //        * valid load funct3 only
    //        * valid store funct3 only
    //        * valid branch funct3 only
    //        * JALR requires funct3==000
    //        * R-type funct7 must be legal
    //        * shift-immediate encodings must be legal
    //        * reject unsupported opcodes/extensions
    //    - Suppress side effects on illegal instructions.
    //
    // 3. Corner cases
    //    - No reset: purely combinational block.
    //    - Illegal encodings must force:
    //        RegWriteD=0, MemRWD=00, BranchD=0, JumpD=0
    //    - Other outputs may safely default.
    //    - For I-type SLT/SLTU, SubArithD=1 per spec.
    //    - For shift-immediates:
    //        SLLI requires funct7=0000000
    //        SRLI requires funct7=0000000
    //        SRAI requires funct7=0100000
    //      all other funct7 values illegal.
    //    - For R-type:
    //        funct7=0000000 legal for all listed ALU ops
    //        funct7=0100000 legal only for SUB and SRA
    //
    // 4. Combinational vs sequential
    //    - All logic is combinational.
    //    - No sequential state, no always_ff, no clock.
    //
    // 5. RTL structure
    //    - Extract opcode/funct3/funct7 fields.
    //    - Set safe defaults.
    //    - Decode by opcode in one always_comb.
    //    - Perform legality checks within each opcode class.
    //    - At end, if illegal, suppress all side effects.
    // -------------------------------------------------------------------------

    logic [6:0] opcode;
    logic [2:0] funct3;
    logic [6:0] funct7;

    assign opcode = InstrD[6:0];
    assign funct3 = InstrD[14:12];
    assign funct7 = InstrD[31:25];

    always_comb begin
        // Safe defaults
        RegWriteD     = 1'b0;
        ImmSrcD       = 3'b000;
        ALUSrcAD      = 1'b0;
        ALUSrcBD      = 1'b0;
        MemRWD        = 2'b00;
        ResultSrcD    = 3'b000;
        BranchD       = 1'b0;
        JumpD         = 1'b0;
        ALUResultSrcD = 1'b0;
        ALUSelectD    = 3'b000;
        SubArithD     = 1'b0;
        IllegalInstrD = 1'b0;

        unique case (opcode)

            // Load
            7'b0000011: begin
                // RV32I loads: LB/LH/LW/LBU/LHU
                if ((funct3 == 3'b000) || // LB
                    (funct3 == 3'b001) || // LH
                    (funct3 == 3'b010) || // LW
                    (funct3 == 3'b100) || // LBU
                    (funct3 == 3'b101)) begin // LHU
                    RegWriteD     = 1'b1;
                    ImmSrcD       = 3'b000;
                    ALUSrcAD      = 1'b0;
                    ALUSrcBD      = 1'b1;
                    MemRWD        = 2'b10;
                    ResultSrcD    = 3'b001;
                    BranchD       = 1'b0;
                    JumpD         = 1'b0;
                    ALUResultSrcD = 1'b0;
                    ALUSelectD    = 3'b000;
                    SubArithD     = 1'b0;
                end
                else begin
                    IllegalInstrD = 1'b1;
                end
            end

            // I-type ALU
            7'b0010011: begin
                RegWriteD     = 1'b1;
                ImmSrcD       = 3'b000;
                ALUSrcAD      = 1'b0;
                ALUSrcBD      = 1'b1;
                MemRWD        = 2'b00;
                ResultSrcD    = 3'b000;
                BranchD       = 1'b0;
                JumpD         = 1'b0;
                ALUResultSrcD = 1'b0;
                ALUSelectD    = funct3;
                SubArithD     = 1'b0;

                unique case (funct3)
                    3'b000: begin // ADDI
                        SubArithD = 1'b0;
                    end
                    3'b001: begin // SLLI
                        if (funct7 == 7'b0000000) begin
                            SubArithD = 1'b0;
                        end
                        else begin
                            IllegalInstrD = 1'b1;
                        end
                    end
                    3'b010: begin // SLTI
                        SubArithD = 1'b1;
                    end
                    3'b011: begin // SLTIU
                        SubArithD = 1'b1;
                    end
                    3'b100: begin // XORI
                        SubArithD = 1'b0;
                    end
                    3'b101: begin // SRLI/SRAI
                        if (funct7 == 7'b0000000) begin
                            SubArithD = 1'b0; // SRLI
                        end
                        else if (funct7 == 7'b0100000) begin
                            SubArithD = 1'b1; // SRAI
                        end
                        else begin
                            IllegalInstrD = 1'b1;
                        end
                    end
                    3'b110: begin // ORI
                        SubArithD = 1'b0;
                    end
                    3'b111: begin // ANDI
                        SubArithD = 1'b0;
                    end
                    default: begin
                        IllegalInstrD = 1'b1;
                    end
                endcase
            end

            // AUIPC
            7'b0010111: begin
                RegWriteD     = 1'b1;
                ImmSrcD       = 3'b100;
                ALUSrcAD      = 1'b1;
                ALUSrcBD      = 1'b1;
                MemRWD        = 2'b00;
                ResultSrcD    = 3'b000;
                BranchD       = 1'b0;
                JumpD         = 1'b0;
                ALUResultSrcD = 1'b0;
                ALUSelectD    = 3'b000;
                SubArithD     = 1'b0;
            end

            // Store
            7'b0100011: begin
                // RV32I stores: SB/SH/SW
                if ((funct3 == 3'b000) || // SB
                    (funct3 == 3'b001) || // SH
                    (funct3 == 3'b010)) begin // SW
                    RegWriteD     = 1'b0;
                    ImmSrcD       = 3'b001;
                    ALUSrcAD      = 1'b0;
                    ALUSrcBD      = 1'b1;
                    MemRWD        = 2'b01;
                    ResultSrcD    = 3'b000;
                    BranchD       = 1'b0;
                    JumpD         = 1'b0;
                    ALUResultSrcD = 1'b0;
                    ALUSelectD    = 3'b000;
                    SubArithD     = 1'b0;
                end
                else begin
                    IllegalInstrD = 1'b1;
                end
            end

            // R-type ALU
            7'b0110011: begin
                RegWriteD     = 1'b1;
                ImmSrcD       = 3'b000;
                ALUSrcAD      = 1'b0;
                ALUSrcBD      = 1'b0;
                MemRWD        = 2'b00;
                ResultSrcD    = 3'b000;
                BranchD       = 1'b0;
                JumpD         = 1'b0;
                ALUResultSrcD = 1'b0;
                ALUSelectD    = funct3;
                SubArithD     = 1'b0;

                if (funct7 == 7'b0000000) begin
                    // Legal base encodings for:
                    // ADD, SLL, SLT, SLTU, XOR, SRL, OR, AND
                    if ((funct3 == 3'b010) || (funct3 == 3'b011)) begin
                        SubArithD = 1'b1; // SLT / SLTU per spec
                    end
                    else begin
                        SubArithD = 1'b0;
                    end
                end
                else if (funct7 == 7'b0100000) begin
                    if (funct3 == 3'b000) begin
                        // SUB
                        SubArithD = 1'b1;
                    end
                    else if (funct3 == 3'b101) begin
                        // SRA
                        SubArithD = 1'b1;
                    end
                    else begin
                        IllegalInstrD = 1'b1;
                    end
                end
                else begin
                    IllegalInstrD = 1'b1;
                end
            end

            // LUI
            7'b0110111: begin
                RegWriteD     = 1'b1;
                ImmSrcD       = 3'b100;
                ALUSrcAD      = 1'b0;
                ALUSrcBD      = 1'b1;
                MemRWD        = 2'b00;
                ResultSrcD    = 3'b000;
                BranchD       = 1'b0;
                JumpD         = 1'b0;
                ALUResultSrcD = 1'b1;
                ALUSelectD    = 3'b000;
                SubArithD     = 1'b0;
            end

            // Branch
            7'b1100011: begin
                // RV32I branches: BEQ/BNE/BLT/BGE/BLTU/BGEU
                if ((funct3 == 3'b000) || // BEQ
                    (funct3 == 3'b001) || // BNE
                    (funct3 == 3'b100) || // BLT
                    (funct3 == 3'b101) || // BGE
                    (funct3 == 3'b110) || // BLTU
                    (funct3 == 3'b111)) begin // BGEU
                    RegWriteD     = 1'b0;
                    ImmSrcD       = 3'b010;
                    ALUSrcAD      = 1'b1;
                    ALUSrcBD      = 1'b1;
                    MemRWD        = 2'b00;
                    ResultSrcD    = 3'b000;
                    BranchD       = 1'b1;
                    JumpD         = 1'b0;
                    ALUResultSrcD = 1'b0;
                    ALUSelectD    = 3'b000;
                    SubArithD     = 1'b0;
                end
                else begin
                    IllegalInstrD = 1'b1;
                end
            end

            // JALR
            7'b1100111: begin
                if (funct3 == 3'b000) begin
                    RegWriteD     = 1'b1;
                    ImmSrcD       = 3'b000;
                    ALUSrcAD      = 1'b0;
                    ALUSrcBD      = 1'b1;
                    MemRWD        = 2'b00;
                    ResultSrcD    = 3'b000;
                    BranchD       = 1'b0;
                    JumpD         = 1'b1;
                    ALUResultSrcD = 1'b1;
                    ALUSelectD    = 3'b000;
                    SubArithD     = 1'b0;
                end
                else begin
                    IllegalInstrD = 1'b1;
                end
            end

            // JAL
            7'b1101111: begin
                RegWriteD     = 1'b1;
                ImmSrcD       = 3'b011;
                ALUSrcAD      = 1'b1;
                ALUSrcBD      = 1'b1;
                MemRWD        = 2'b00;
                ResultSrcD    = 3'b000;
                BranchD       = 1'b0;
                JumpD         = 1'b1;
                ALUResultSrcD = 1'b1;
                ALUSelectD    = 3'b000;
                SubArithD     = 1'b0;
            end

            // Everything else is unsupported / illegal in this decoder:
            // FENCE/FENCE.I, SYSTEM (ECALL/EBREAK/CSR), misc extensions, etc.
            default: begin
                IllegalInstrD = 1'b1;
            end
        endcase

        // Suppress architectural side effects on illegal instructions.
        if (IllegalInstrD) begin
            RegWriteD = 1'b0;
            MemRWD    = 2'b00;
            BranchD   = 1'b0;
            JumpD     = 1'b0;
        end
    end

endmodule
```
