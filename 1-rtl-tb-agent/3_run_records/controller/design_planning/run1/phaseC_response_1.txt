    output logic [1:0]  MemRWD,

                if ((funct3 == 3'b000) ||

                end

                ImmSrcD       = 3'b000;
                ALUSrcAD      = 1'b0;

                        SubArithD = 1'b0;
                    end

                            SubArithD = 1'b0;
                        end

                            IllegalInstrD = 1'b1;
                        end

                            SubArithD = 1'b0;
                        end
                        else if (funct7 == 7'b0100000) begin
                            SubArithD = 1'b1;
                        end
                        else begin
                            IllegalInstrD = 1'b1;

                        SubArithD = 1'b0;
                    end
                    3'b111: begin
                        SubArithD = 1'b0;
                    end

                        IllegalInstrD = 1'b1;

                ImmSrcD       = 3'b100;
                ALUSrcAD      = 1'b1;
                ALUSrcBD      = 1'b1;
                ResultSrcD    = 3'b000;
                ALUResultSrcD = 1'b0;
                ALUSelectD    = 3'b000;

                ALUSrcBD      = 1'b0;
                ResultSrcD    = 3'b000;

                ALUSelectD    = funct3;

                    ImmSrcD       = 3'b000;

                    ALUSrcBD      = 1'b1;
                    ResultSrcD    = 3'b000;
                    JumpD         = 1'b1;
                    ALUResultSrcD = 1'b1;

                ALUSelectD    = 3'b000;

        if (IllegalInstrD) begin
            RegWriteD     = 1'b0;
            MemRWD        = 2'b00;
            BranchD       = 1'b0;
            JumpD         = 1'b0;
            ImmSrcD       = 3'b000;

            ResultSrcD    = 3'b000;
            ALUResultSrcD = 1'b0;
            ALUSelectD    = 3'b000;
            SubArithD     = 1'b0;
        end
    end

endmodule
